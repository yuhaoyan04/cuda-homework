cmake_minimum_required(VERSION 3.21)
project(CudaSDFAssignment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(sdf_common src/sdf_common.cpp)
target_include_directories(sdf_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_library(sdf_cpu src/sdfgen_style.cpp)
target_link_libraries(sdf_cpu PUBLIC sdf_common)
add_executable(sdfbench_cpu src/cpu_main.cpp)
target_link_libraries(sdfbench_cpu PRIVATE sdf_cpu)

option(BUILD_CUDA "Build the GPU student placeholder and timing harness" OFF)
if(BUILD_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120) # RTX 5090 / Blackwell, with a sufficiently recent CUDA toolkit
  endif()
  add_executable(sdfbench_gpu src/gpu_main.cu student/student_sdf.cu)
  target_link_libraries(sdfbench_gpu PRIVATE sdf_common)
  target_include_directories(sdfbench_gpu PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/student)
endif()
